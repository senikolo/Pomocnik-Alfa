// Geometry of the map and a schematic stop sequence; no fabricated live movement.
function busGeoPoint(latitude,longitude,level){
 var lat=Number(latitude),lon=Number(longitude);
 if(!Number.isFinite(lat)||!Number.isFinite(lon)||Math.abs(lat)>90||Math.abs(lon)>180)return null;
 lat=Math.max(-85.05112878,Math.min(85.05112878,lat));
 var n=Math.pow(2,level),r=lat*Math.PI/180;
 return {x:(lon+180)/360*n,y:(1-Math.log(Math.tan(r)+1/Math.cos(r))/Math.PI)/2*n};
}
function busMapLayout(vehicle,width,height,zoom){
 var level=Math.ceil(zoom),size=256*Math.pow(2,zoom-level),n=Math.pow(2,level),center=busGeoPoint(vehicle.latitude,vehicle.longitude,level);
 if(!center)return null;
 function locate(lat,lon){var p=busGeoPoint(lat,lon,level);if(!p)return null;var dx=p.x-center.x;if(dx>n/2)dx-=n;if(dx<-n/2)dx+=n;return {x:width/2+dx*size,y:height/2+(p.y-center.y)*size};}
 var tiles=[],minX=Math.floor(center.x-width/2/size),maxX=Math.floor(center.x+width/2/size),minY=Math.floor(center.y-height/2/size),maxY=Math.floor(center.y+height/2/size);
 for(var x=minX;x<=maxX;x++)for(var y=minY;y<=maxY;y++)if(y>=0&&y<n)tiles.push({x:((x%n+n)%n),y:y,left:width/2+(x-center.x)*size,top:height/2+(y-center.y)*size,level:level,size:size});
 var stops=(vehicle.stops||[]).map(function(st){var p=st.latitude===null||st.longitude===null?null:locate(st.latitude,st.longitude);return p?Object.assign({},st,p):null;}).filter(function(st){return st&&st.x>=8&&st.x<=width-8&&st.y>=8&&st.y<=height-24;});
 stops.sort(function(a,b){return Number(b.index===vehicle.boardingIndex)-Number(a.index===vehicle.boardingIndex)||Number(b.index===vehicle.currentIndex)-Number(a.index===vehicle.currentIndex)||Math.hypot(a.x-width/2,a.y-height/2)-Math.hypot(b.x-width/2,b.y-height/2);});
 var next=(vehicle.stops||[])[vehicle.currentIndex+(vehicle.atStop?1:0)],direction=next&&next.latitude!==null&&next.longitude!==null?locate(next.latitude,next.longitude):null;
 if(direction&&Math.hypot(direction.x-width/2,direction.y-height/2)<3)direction=null;
 return {tiles:tiles,stops:stops.slice(0,8),direction:direction,tileSize:size,level:level};
}
function busAxisRows(vehicle,full){
 var stops=vehicle.stops||[],current=vehicle.currentIndex,boarding=vehicle.boardingIndex;
 if(!stops.length||!Number.isInteger(current)||current<0||current>=stops.length)return [];
 var indexes=full?stops.map(function(_,i){return i;}):[current-1,current,current+1,boarding].filter(function(i,p,a){return i>=0&&i<stops.length&&a.indexOf(i)===p;}).sort(function(a,b){return a-b;});
 var rows=[],previous=-1;
 indexes.forEach(function(index){
  if(previous>=0&&index-previous>1)rows.push({type:'gap',count:index-previous-1});
  if(index===current&&!vehicle.atStop)rows.push({type:'bus',name:'Autobus teraz',detail:vehicle.positionLabel+': '+stops[current].name});
  rows.push({type:'stop',name:stops[index].name,index:index,boarding:index===boarding,current:index===current,bus:index===current&&vehicle.atStop,past:index<current});
  previous=index;
 });
 return rows;
}

 async function busLookUp(){
  var b=state.bus,provider=busData();if(!b||!b.line||!b.direction||!b.stop)return;
  if(busNormalize(b.stop)==='ispina')b.stop='Ispina Most';
  clearChoices();var job=begin();status.textContent='Sprawdzam najbliższy odjazd…';
  try{
    if(!provider||typeof provider.busArrivals!=='function')throw Error('NO_SCHEDULE_PROVIDER');
    var arrival=provider.busArrivals(b.line,b.direction,b.stop,Date.now());
    if(job.id!==session)return;
    var scheduledTrips=(arrival&&arrival.items||[]).filter(function(x){return x&&x.tripId;});
    var upcomingIds=scheduledTrips.map(function(x){return String(x.tripId);});
    say(busFormatArrivals(arrival,b));
    if(b.line==='GMINNA'){say('Linia gminna: nie mam potwierdzonego źródła GPS. Godzina powyżej pochodzi z rozkładu.');completed('✓ Sprawdzono rozkład');return;}
    if(!upcomingIds.length){say('Nie mogę przypisać pozycji GPS do konkretnego najbliższego kursu. Podałem jedynie dostępny rozkład.');completed('✓ Sprawdzono rozkład');return;}
    status.textContent='Sprawdzam dokładną pozycję GPS…';showTyping();
    var feed=null;
    try{if(w.IspinaNative&&typeof w.IspinaNative.getMldVehiclePositionsJson==='function'&&typeof w.IspinaNative.requestMldVehiclePositions==='function'){
      feed=await busReadNative(job);
    }else if(w.IspinaNative&&typeof w.IspinaNative.getMldVehiclePositionsJson==='function'){
      var raw=w.IspinaNative.getMldVehiclePositionsJson();if(raw)feed=JSON.parse(raw);
    }}catch(e){}
    if(!feed&&w.fetch){var controller=w.AbortController?new w.AbortController():null;
      var timer=setTimeout(function(){if(controller)controller.abort();},3500);
      try{var response=await w.fetch(BUS_FEED,{method:'GET',mode:'cors',cache:'no-store',credentials:'omit',signal:controller?controller.signal:job.signal});if(response.ok)feed=await response.json();}catch(e){}finally{clearTimeout(timer);}
    }
    if(job.id!==session)return;
    if(feed){var result=provider.busVehicle(feed,b.line,b.direction,b.stop,upcomingIds);
      if(result.status==='live'&&result.vehicles.length){result.vehicles.slice(0,1).forEach(function(v){
        var trip=scheduledTrips.filter(function(x){return String(x.tripId)===v.tripId;})[0];

        var card=say(b.line+' → '+b.direction+'\n'+v.positionLabel+': '+v.place+'\nTwój przystanek: '+b.stop+'\nOdjazd według '+(trip&&trip.live?'danych LIVE':'rozkładu')+': '+when+'\nPomiar GPS: '+new Date(v.at).toLocaleDateString('pl-PL')+' '+new Date(v.at).toLocaleTimeString('pl-PL',{hour:'2-digit',minute:'2-digit',second:'2-digit'}));
        card.classList.add('pil-gps-card');card.insertBefore(el('span','pil-gps-badge','POTWIERDZONY GPS'),card.firstChild);busAxis(card,v);busMap(card,v);
      });completed('✓ Pozycja GPS potwierdzona');}
      else{say('Nie mam świeżej pozycji GPS przypisanej do tych konkretnych kursów. Podane wyżej godziny pozostają rozkładowe, o ile nie oznaczyłem ich LIVE.');completed('✓ Sprawdzono odjazdy');}
    }else{say('Nie udało się połączyć ze źródłem GPS MLD. Powyżej są godziny odjazdów z rozkładu, z osobnym oznaczeniem świeżego opóźnienia, jeśli jest dostępne. Nie potwierdzam pozycji pojazdu.');completed('✓ Sprawdzono odjazdy');}
  }catch(e){if(job.id===session){say('Nie mogę teraz odczytać rozkładu dla '+b.stop+'. Otwórz Odjazdy i odśwież dane. Potem wróć tutaj — zapamiętam linię, kierunek i przystanek.');completed('! Brak danych odjazdów');}}
  finally{if(job.id===session){hideTyping();active=null;cancelButton.hidden=true;
    tile('↻ Sprawdź ponownie',busLookUp);
    tile('Zmień przystanek',function(){b.stop=null;b.stage='stop';busPrompt();});
    tile('Zmień kierunek',function(){b.direction=null;b.stop=null;b.stage='direction';busPrompt();});
    tile('Inna linia',function(){state.bus={line:null,direction:null,stop:null,stage:'line'};busPrompt();});
    persist();log.scrollTop=log.scrollHeight;
  }}
